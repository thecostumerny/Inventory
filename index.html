/** * Web App entry point and server functions for the Inventory app. * Lives in the same Apps Script project as Code.gs. */ const LOCATIONS = ['FC', 'MH', 'BA']; const ZONES_SHEET = 'Zones'; // Which columns of the Zones sheet hold each location's zones (1-based). // A location with several columns gets all of them in one list. // A location not listed here skips the zone step. const ZONE_COLUMNS = { FC: [1], // A BA: [5, 6] // E and F }; // Each save is also logged to " Scans", e.g. "FC Scans". const SCAN_LOG_HEADERS = ['SKU', 'Description', 'Counted', 'Zone', 'Counted By']; // Columns on FC / MH / BA (0-based: A = 0), as built by Code.gs const TITLE_IDX = 0; // A const VARIANT_IDXS = [1, 2, 3]; // B, C, D const SKU_IDX = 4; // E const BARCODE_IDX = 5; // F (one or more barcodes, comma separated) const PRICE_IDX = 6; // G const EXPECTED_IDX = 8; // I const COUNT_COL = 10; // J (1-based, for writing) const READ_COLS = 10; // read A–J /** Serves the app page when the Web App URL is opened. */ function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Inventory') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /** * Data requests from the app when it runs outside Apps Script (e.g. GitHub Pages). * Body: {action, args, pass}. Only these actions can be called, and each one * checks the password itself. */ const API_ACTIONS = { ping: function () { return JSON.stringify(ping()); }, login: function (args, pass) { return JSON.stringify(login(pass)); }, getAppData: function (args, pass) { return JSON.stringify(getAppData(pass)); }, getItems: function (args, pass) { return getItems(args[0], pass); }, // already JSON saveCounts: function (args, pass) { return JSON.stringify(saveCounts(args[0] || [], pass)); } }; function doPost(e) { let out; try { const body = JSON.parse(e.postData.contents); const fn = API_ACTIONS[body.action]; if (!fn) throw new Error('Unknown action'); out = '{"ok":true,"data":' + fn(body.args || [], body.pass) + '}'; } catch (err) { out = JSON.stringify({ ok: false, error: err.message }); } return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON); } /* ---------- password ---------- * Any value in column A of the "Login" sheet is a valid password. * The list is cached for 60 seconds so checks stay fast; * changes on the sheet take effect within a minute. */ const LOGIN_SHEET = 'Login'; /** Quick password check for the login screen. */ function login(pass) { checkLogin_(pass); return true; } /** * Called when the login screen opens, before anyone types. It wakes the * script up and loads the password list into the cache, so the real check * is fast. Returns nothing about the passwords. */ function ping() { try { getPasswords_(); } catch (e) {} return true; } function checkLogin_(pass) { pass = String(pass == null ? '' : pass).trim(); if (!pass || getPasswords_().indexOf(pass) === -1) { throw new Error('LOGIN_REQUIRED'); } } function getPasswords_() { const cache = CacheService.getScriptCache(); const hit = cache.get('login_list'); if (hit) return JSON.parse(hit); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheets().filter(function (sh) { return sh.getName().toLowerCase() === LOGIN_SHEET.toLowerCase(); })[0]; if (!sheet) throw new Error('Login is not set up yet.'); const lastRow = sheet.getLastRow(); const list = lastRow < 1 ? [] : sheet.getRange(1, 1, lastRow, 1).getDisplayValues() .map(function (r) { return String(r[0]).trim(); }) .filter(String); if (!list.length) throw new Error('Login is not set up yet.'); cache.put('login_list', JSON.stringify(list), 60); return list; } /** * Called once in the background as soon as the page opens. * Returns everything the first screens need for every location, * so picking a location and zone is instant. * @return {Object} */ function getAppData(pass) { checkLogin_(pass); const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = {}; LOCATIONS.forEach(function (loc) { const sheet = ss.getSheetByName(loc); if (!sheet) { data[loc] = { error: loc + ' is not ready for counting yet.' }; return; } try { data[loc] = { location: loc, itemCount: Math.max(sheet.getLastRow() - 1, 0), // minus header row zones: getZones_(loc) }; } catch (e) { data[loc] = { error: e.message }; } }); return data; } /** * Loads the item list for a location so scans can be looked up on the device. * * Each row sends only what the app shows: SKU, name (Title + variants), * Price and Expected, plus the barcode list and current Count. * Built for speed with ~15,000 rows: one read of A:J, returned as JSON. * * @param {string} location 'FC', 'MH' or 'BA' * @return {string} JSON: {rows: [sku, name, price, expected][], codes, counts} * codes[i] is the raw barcode cell (may be "a, b, c") */ function getItems(location, pass) { checkLogin_(pass); if (LOCATIONS.indexOf(location) === -1) { throw new Error('Unknown location: ' + location); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(location); if (!sheet) throw new Error(location + ' is not ready for counting yet.'); const lastRow = sheet.getLastRow(); if (lastRow < 2) return JSON.stringify({ rows: [], codes: [], counts: [] }); const vals = sheet.getRange(2, 1, lastRow - 1, READ_COLS).getValues(); const rows = new Array(vals.length); const codes = new Array(vals.length); const counts = new Array(vals.length); for (let r = 0; r < vals.length; r++) { const v = vals[r]; codes[r] = String(v[BARCODE_IDX]).trim(); counts[r] = v[COUNT_COL - 1] === '' ? '' : String(v[COUNT_COL - 1]); const price = v[PRICE_IDX]; rows[r] = [ String(v[SKU_IDX]), itemName_(v), typeof price === 'number' ? '$' + price.toFixed(2) : String(price), String(v[EXPECTED_IDX]) ]; } return JSON.stringify({ rows: rows, codes: codes, counts: counts }); } /** Title followed by any variants, skipping blanks and "Default Title". */ function itemName_(v) { const parts = [String(v[TITLE_IDX]).trim()]; VARIANT_IDXS.forEach(function (i) { const x = String(v[i]).trim(); if (x && x.toLowerCase() !== 'default title') parts.push(x); }); return parts.filter(String).join(' '); } /** * Saves a batch of counts in ONE call (the app sends whatever has queued up * while the previous save was running, so fast scanning stays fast). * * For each entry: adds qty to column J (Count) of the item's row and logs * a row to " Scans". Counts are added, not replaced. * A lock keeps two devices saving at once from overwriting each other. * * Each entry gets its own result, so one bad entry doesn't block the rest. * * @param {Array<{loc: string, index: number, sku: string, code: string, * qty: number, zone: string, initials: string}>} entries * @return {Array<{total: number}|{error: string}>} one result per entry, same order */ function saveCounts(entries, pass) { checkLogin_(pass); const results = new Array(entries.length); const ss = SpreadsheetApp.getActiveSpreadsheet(); const lock = LockService.getScriptLock(); lock.waitLock(30000); try { // Barcode links (found by SKU after a scan wasn't found) go to "Barcode Update" const linkIdxs = []; entries.forEach(function (e, i) { if (e.type === 'link') linkIdxs.push(i); }); if (linkIdxs.length) saveBarcodeLinks_(entries, linkIdxs, results); // Group count entries by location const byLoc = {}; entries.forEach(function (e, i) { if (e.type === 'link') return; (byLoc[e.loc] = byLoc[e.loc] || []).push(i); }); Object.keys(byLoc).forEach(function (loc) { const idxs = byLoc[loc]; const failAll = function (msg) { idxs.forEach(function (i) { results[i] = { error: msg }; }); }; if (LOCATIONS.indexOf(loc) === -1) return failAll('Unknown location: ' + loc); const sheet = ss.getSheetByName(loc); if (!sheet) return failAll(loc + ' is not ready for counting yet.'); const lastRow = sheet.getLastRow(); let allSkus = null; // column E, read only if a row has moved const totals = {}; // row number -> new total (for repeats in this batch) const logRows = []; idxs.forEach(function (i) { const e = entries[i]; try { const qty = Number(e.qty); if (!isFinite(qty) || qty < 0 || Math.floor(qty) !== qty) { throw new Error('Quantity must be a whole number.'); } const skuKey = normCode_(e.sku); let row = Number(e.index) + 2; // One read gives the SKU check, name and current Count let vals = row <= lastRow ? sheet.getRange(row, 1, 1, READ_COLS).getDisplayValues()[0] : null; // If the sheet was rebuilt since the app loaded, find the SKU again if (!vals || normCode_(vals[SKU_IDX]) !== skuKey) { if (!allSkus) { allSkus = sheet.getRange(2, SKU_IDX + 1, Math.max(lastRow - 1, 1), 1).getValues() .map(function (r) { return normCode_(r[0]); }); } const found = allSkus.indexOf(skuKey); if (found === -1) throw new Error('SKU ' + e.sku + ' is no longer in ' + loc + '.'); row = found + 2; vals = sheet.getRange(row, 1, 1, READ_COLS).getDisplayValues()[0]; } const current = (row in totals) ? totals[row] : (Number(String(vals[COUNT_COL - 1]).replace(/,/g, '')) || 0); const total = current + qty; totals[row] = total; logRows.push([vals[SKU_IDX], itemName_(vals), qty, e.zone || '', e.initials || '']); results[i] = { total: total }; } catch (err) { results[i] = { error: err.message }; } }); // Write the new totals to column J Object.keys(totals).forEach(function (row) { sheet.getRange(Number(row), COUNT_COL).setValue(totals[row]); }); // Write all log rows in one go if (logRows.length) { const log = getScanLog_(loc); log.getRange(log.getLastRow() + 1, 1, logRows.length, logRows[0].length) .setValues(logRows); } }); return results; } finally { lock.releaseLock(); } } /* * "Barcode Update" sheet: SKU and the barcode that was scanned but not found, * so the barcode can be added to that SKU in the Data sheet later. * Columns: A SKU, B Barcode, C Location, D Added By, E Date. * A SKU + barcode pair already on the sheet isn't added twice. */ const BARCODE_UPDATE_SHEET = 'Barcode Update'; const BARCODE_UPDATE_HEADERS = ['SKU', 'Barcode', 'Location', 'Added By', 'Date']; function saveBarcodeLinks_(entries, idxs, results) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(BARCODE_UPDATE_SHEET); if (!sheet) { sheet = ss.insertSheet(BARCODE_UPDATE_SHEET); sheet.getRange(1, 1, 1, BARCODE_UPDATE_HEADERS.length) .setValues([BARCODE_UPDATE_HEADERS]).setFontWeight('bold'); sheet.setFrozenRows(1); sheet.getRange('A:B').setNumberFormat('@'); // keep SKUs and barcodes exactly as-is } // Pairs already recorded const seen = {}; const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 2).getDisplayValues().forEach(function (r) { seen[normCode_(r[0]) + '|' + normCode_(r[1])] = true; }); } const rows = []; const now = new Date(); idxs.forEach(function (i) { const e = entries[i]; try { if (!e.sku || !e.code) throw new Error('Missing SKU or barcode.'); const key = normCode_(e.sku) + '|' + normCode_(e.code); if (!seen[key]) { seen[key] = true; rows.push([String(e.sku), String(e.code), e.loc || '', e.initials || '', now]); } results[i] = { ok: true }; } catch (err) { results[i] = { error: err.message }; } }); if (rows.length) { const start = sheet.getLastRow() + 1; sheet.getRange(start, 1, rows.length, 2).setNumberFormat('@'); sheet.getRange(start, 1, rows.length, rows[0].length).setValues(rows); } } /** Gets the " Scans" sheet, creating it with headers if needed. */ function getScanLog_(location) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const name = location + ' Scans'; let log = ss.getSheetByName(name); if (!log) { log = ss.insertSheet(name); const n = SCAN_LOG_HEADERS.length; log.getRange(1, 1, 1, n).setValues([SCAN_LOG_HEADERS]).setFontWeight('bold'); log.setFrozenRows(1); log.getRange('A:A').setNumberFormat('@'); // keep SKUs as text (leading zeros) } return log; } /** Same barcode matching as the app: case-insensitive, numeric codes ignore leading zeros. */ function normCode_(c) { c = String(c).trim().toUpperCase(); if (/^\d+$/.test(c)) c = c.replace(/^0+(?=\d)/, ''); return c; } /** * Reads the zone list for a location from the Zones sheet. * Row 1 is the header and is skipped; blanks and duplicates are skipped too. * @return {string[]|null} null when the location has no zone step. */ function getZones_(location) { const cols = ZONE_COLUMNS[location]; if (!cols) return null; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ZONES_SHEET); if (!sheet) { throw new Error('Zones are not set up yet.'); } const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; // One read covering all of this location's columns const first = Math.min.apply(null, cols); const last = Math.max.apply(null, cols); const raw = sheet.getRange(2, first, lastRow - 1, last - first + 1).getDisplayValues(); const seen = {}; const zones = []; cols.forEach(function (c) { // column by column, top to bottom raw.forEach(function (row) { const z = String(row[c - first]).trim(); if (!z) return; const key = z.toUpperCase(); if (seen[key]) return; seen[key] = true; zones.push(z); }); }); return zones; }
